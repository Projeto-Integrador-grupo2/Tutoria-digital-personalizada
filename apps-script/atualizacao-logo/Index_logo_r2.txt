    <script>
      (function () {
      'use strict';
      const $ = selector => document.querySelector(selector);
      const state = { busy: false, config: null, auth: null, session: null, pending: null, profiles: [] };
      const levels = ['Diagnóstico', 'Pergunta orientadora', 'Pista conceitual', 'Microetapas', 'Exemplo semelhante', 'Correção comentada'];
      const hasHost = () => Boolean(window.google && google.script && google.script.run);

      function host(method, ...args) {
        return new Promise((resolve, reject) => {
          const timer = setTimeout(() => reject(new Error('O envio demorou mais que o esperado. Reenvie a mesma mensagem para consultar o resultado.')), 55000);
          google.script.run.withSuccessHandler(value => { clearTimeout(timer); resolve(value); })
            .withFailureHandler(() => { clearTimeout(timer); reject(new Error('Não foi possível conectar ao serviço. Tente novamente.')); })[method](...args);
        });
      }

      async function accessToken() {
        if (!state.auth) throw new Error('Entre na sua conta para continuar.');
        if (Date.now() > state.auth.expiresAt - 60000) {
          const response = await fetch(state.config.supabaseUrl + '/auth/v1/token?grant_type=refresh_token', {
            method: 'POST', headers: { apikey: state.config.publicKey, 'Content-Type': 'application/json' },
            body: JSON.stringify({ refresh_token: state.auth.refresh_token }), signal: AbortSignal.timeout(15000)
          });
          const data = await response.json();
          if (!response.ok) { clearAccount(); throw new Error('Seu acesso expirou. Entre novamente.'); }
          setAuth(data);
        }
        return state.auth.access_token;
      }

      function setAuth(data) {
        state.auth = { access_token: data.access_token, refresh_token: data.refresh_token, expiresAt: Date.now() + data.expires_in * 1000 };
        $('#authForm').hidden = true;
        $('#accountActions').hidden = false;
        $('#userChip').textContent = 'Conta conectada';
      }

      async function api(body) {
        const token = await accessToken();
        const data = await host('tutorApi', body, token);
        if (data.error) {
          const error = new Error(data.error.message + (data.error.request_id ? ' Código de suporte: ' + data.error.request_id : ''));
          error.code = data.error.code;
          if (error.code === 'invalid_or_expired_token') clearAccount();
          throw error;
        }
        return data;
      }

      function append(label, text, role = 'assistant') {
        const wrapper = document.createElement('div');
        wrapper.className = 'message ' + role;
        const heading = document.createElement('div');
        heading.className = 'label'; heading.textContent = label;
        const bubble = document.createElement('div');
        bubble.className = 'bubble'; bubble.textContent = text;
        wrapper.append(heading, bubble); $('#messages').append(wrapper);
        $('#messages').scrollTop = $('#messages').scrollHeight;
      }

      async function run(action) {
        if (state.busy) return;
        state.busy = true;
        document.querySelectorAll('button, #profile').forEach(item => { item.disabled = true; });
        $('#messages').setAttribute('aria-busy', 'true');
        try { await action(); }
        catch (error) { append('Sistema', error.message || 'Não foi possível concluir a operação.'); }
        finally {
          state.busy = false;
          document.querySelectorAll('button, #profile').forEach(item => { item.disabled = false; });
          $('#messages').setAttribute('aria-busy', 'false');
        }
      }

      function clearAccount() {
        state.auth = null; state.session = null; state.pending = null;
        $('#authForm').hidden = false; $('#accountActions').hidden = true;
        $('#historyList').replaceChildren(); $('#historyList').hidden = true;
        $('#messages').replaceChildren(); $('#message').value = '';
        $('#activity').value = ''; $('#activity').disabled = false; $('#subject').disabled = false;
        $('#privacyAccepted').checked = false;
        ['reflection', 'remainingDifficulty', 'nextStep'].forEach(id => { $('#' + id).value = ''; });
        $('#userChip').textContent = 'Acesso encerrado';
      }

      function sessionDisplay(session) {
        state.session = session;
        $('#subject').value = session.subject;
        $('#activity').value = session.activity;
        $('#profile').value = session.profile;
        $('#subject').disabled = true; $('#activity').disabled = true;
        $('#hintStatus').textContent = levels[session.hint_level] || levels[0];
        updateProfile();
      }

      async function send() {
        if (!$('#setupForm').reportValidity() || !$('#messageForm').reportValidity()) return;
        if (!hasHost()) { append('Sistema', 'Prévia local. Publique a interface no Apps Script para entrar e estudar.'); return; }
        if (!state.session) {
          const data = await api({ action: 'create_session', subject: $('#subject').value, activity: $('#activity').value.trim(),
            profile: $('#profile').value, consent_version: '1.2', privacy_notice_accepted: $('#privacyAccepted').checked });
          sessionDisplay(data.session);
        }
        if (state.session.status !== 'active') throw new Error('Esta sessão foi encerrada. Inicie uma nova sessão para continuar.');
        const content = $('#message').value.trim();
        const requestMoreHelp = $('#moreHelp').checked;
        if (!state.pending || state.pending.content !== content || state.pending.request_more_help !== requestMoreHelp) {
          state.pending = { action: 'send_message', session_id: state.session.id, client_request_id: crypto.randomUUID(),
            content, request_more_help: requestMoreHelp };
        }
        try {
          const data = await api(state.pending);
          append('Você', content, 'user');
          append('Tutor', data.tutor.message);
          if (data.availability && data.availability.degraded) append('Sistema', 'A IA está temporariamente indisponível. Esta orientação é uma pista geral para manter seu estudo.');
          state.session.hint_level = data.session.hint_level;
          $('#hintStatus').textContent = levels[data.session.hint_level];
          $('#message').value = ''; $('#moreHelp').checked = false; state.pending = null;
        } catch (error) {
          if (error.code === 'session_changed') {
            await loadSession(state.session.id);
            state.pending = null;
          }
          throw error;
        }
      }

      async function loadSession(id) {
        const data = await api({ action: 'get_session', session_id: id, limit: 50 });
        sessionDisplay(data.session);
        state.pending = null; $('#messages').replaceChildren();
        // Páginas antigas são inseridas antes da página atual, mantendo a leitura cronológica.
        const render = items => items.forEach(item => append(item.role === 'student' ? 'Você' : 'Tutor', item.content, item.role === 'student' ? 'user' : 'assistant'));
        render(data.messages);
        if (data.pagination.has_more) {
          const older = document.createElement('button');
          older.type = 'button'; older.textContent = 'Carregar mensagens anteriores';
          let cursor = data.pagination.next_before_message_id;
          older.addEventListener('click', () => run(async () => {
            const page = await api({ action: 'get_session', session_id: id, limit: 50, before_message_id: cursor });
            const existing = Array.from($('#messages').children).filter(child => child !== older);
            $('#messages').replaceChildren(older); render(page.messages); existing.forEach(child => $('#messages').append(child));
            cursor = page.pagination.next_before_message_id;
            if (!page.pagination.has_more) older.remove();
          }));
          $('#messages').prepend(older);
        }
        if (data.session.summary) append('Resumo de aprendizagem', data.session.summary);
      }

      async function history() {
        const list = $('#historyList'); list.hidden = false; list.replaceChildren();
        async function page(cursor = {}) {
          const data = await api({ action: 'list_sessions', limit: 20, ...cursor });
          if (!data.sessions.length && !list.children.length) list.textContent = 'Você ainda não tem sessões.';
          data.sessions.forEach(session => {
            const button = document.createElement('button'); button.type = 'button'; button.className = 'secondary';
            button.textContent = session.subject + ' · ' + new Date(session.created_at).toLocaleString('pt-BR') + ' · ' + (session.status === 'active' ? 'em estudo' : 'encerrada');
            button.addEventListener('click', () => run(() => loadSession(session.id))); list.append(button);
          });
          if (data.pagination.has_more) {
            const more = document.createElement('button'); more.type = 'button'; more.textContent = 'Mais sessões';
            more.addEventListener('click', () => run(async () => {
              await page({ before_created_at: data.pagination.next_before_created_at, before_session_id: data.pagination.next_before_session_id }); more.remove();
            })); list.append(more);
          }
        }
        await page();
      }

      function updateProfile() {
        const selected = state.profiles.find(item => item.id === $('#profile').value);
        const text = selected ? selected.label + ': ' + selected.description : '';
        $('#profileNote').textContent = text; $('#asideProfile').textContent = text;
      }

      function freshSession() {
        state.session = null; state.pending = null; $('#messages').replaceChildren();
        $('#activity').disabled = false; $('#subject').disabled = false; $('#activity').value = '';
        $('#hintStatus').textContent = levels[0]; $('#message').value = '';
        ['reflection', 'remainingDifficulty', 'nextStep'].forEach(id => { $('#' + id).value = ''; });
        ['usefulness', 'clarity', 'profileFit', 'autonomy'].forEach(id => { $('#' + id).value = ''; });
        $('#activity').focus();
      }

      $('#authForm').addEventListener('submit', event => { event.preventDefault(); run(async () => {
        if (!state.config.configured) throw new Error('A conexão ainda precisa ser configurada no Apps Script.');
        try {
          const response = await fetch(state.config.supabaseUrl + '/auth/v1/token?grant_type=password', {
            method: 'POST', headers: { apikey: state.config.publicKey, 'Content-Type': 'application/json' },
            body: JSON.stringify({ email: $('#email').value.trim(), password: $('#password').value }), signal: AbortSignal.timeout(15000)
          });
          const data = await response.json();
          if (!response.ok) throw new Error('Confira seu acesso ou aguarde antes de tentar novamente.');
          setAuth(data);
          const privacy = await api({ action: 'get_privacy_status' });
          append('Sistema', 'Histórico privado por conta. Prazo de retenção: ' + privacy.privacy.retention_days + ' dias. Não envie dados pessoais na atividade.');
        } finally { $('#password').value = ''; }
      }); });
      $('#messageForm').addEventListener('submit', event => { event.preventDefault(); run(send); });
      $('#setupForm').addEventListener('submit', event => event.preventDefault());
      $('#profile').addEventListener('change', () => { updateProfile(); if (state.session) run(async () => {
        try { const data = await api({ action: 'update_profile', session_id: state.session.id, profile: $('#profile').value }); sessionDisplay(data.session); }
        catch (error) { $('#profile').value = state.session.profile; updateProfile(); throw error; }
      }); });
      $('#summaryButton').addEventListener('click', () => run(async () => {
        if (!state.session) throw new Error('Inicie uma sessão antes de encerrar.');
        const data = await api({ action: 'finish_session', session_id: state.session.id,
          reflection: $('#reflection').value.trim(), remaining_difficulty: $('#remainingDifficulty').value.trim(), next_step: $('#nextStep').value.trim() });
        sessionDisplay(data.session); append('Resumo de aprendizagem', data.session.summary);
      }));
      $('#evaluationButton').addEventListener('click', () => run(async () => {
        if (!state.session) throw new Error('Inicie uma sessão antes de avaliar.');
        const values = ['usefulness', 'clarity', 'profileFit', 'autonomy'].map(id => Number($('#' + id).value));
        if (values.some(value => value < 1 || value > 5)) throw new Error('Para enviar a avaliação opcional, selecione as quatro notas.');
        await api({ action: 'submit_feedback', session_id: state.session.id, useful: values[0], clear: values[1], profile_fit: values[2], autonomy: values[3] });
        append('Sistema', 'Avaliação registrada. Obrigado por participar.');
      }));
      $('#historyButton').addEventListener('click', () => run(history));
      $('#newSessionButton').addEventListener('click', () => { if (!state.busy) freshSession(); });
      $('#deleteHistoryButton').addEventListener('click', () => {
        if (!confirm('Apagar todas as suas sessões, mensagens e avaliações? Esta ação não pode ser desfeita.')) return;
        run(async () => {
          await api({ action: 'delete_all_history', confirmation: 'APAGAR_MEU_HISTORICO' });
          freshSession(); $('#historyList').replaceChildren(); append('Sistema', 'Seu histórico foi apagado.');
        });
      });
      $('#logoutButton').addEventListener('click', () => run(async () => {
        try {
          const token = await accessToken();
          await fetch(state.config.supabaseUrl + '/auth/v1/logout?scope=local', {
            method: 'POST', headers: { apikey: state.config.publicKey, Authorization: 'Bearer ' + token }, signal: AbortSignal.timeout(10000)
          });
        } finally { clearAccount(); }
      }));

      async function boot() {
        const logo = $('#brandLogo');
        // No Apps Script, o src já chega preenchido pelo servidor. Na prévia, usamos o PNG local.
        if (logo && !logo.getAttribute('src').startsWith('data:image/')) logo.src = 'tutoria-digital-logo.png';
        const fallback = { configured: false, privacyNotice: 'Prévia local: entre no endereço publicado para usar o tutor.',
          subjects: ['Matemática', 'Português', 'Física', 'Informática', 'Outra disciplina'],
          profiles: [{ id: 'direct', label: 'Direto', description: 'Uma ação por vez.' }, { id: 'supportive', label: 'Acolhedor', description: 'Apoio com ritmo cuidadoso.' }, { id: 'investigative', label: 'Investigativo', description: 'Investigue e justifique suas ideias.' }] };
        function hydrate(config) {
          state.config = config;
          state.profiles = config.profiles;
          for (const [id, items] of [['subject', config.subjects.map(item => ({ id: item, label: item }))], ['profile', config.profiles]]) {
            $('#' + id).replaceChildren();
            items.forEach(item => { const option = document.createElement('option'); option.value = item.id; option.textContent = item.label; $('#' + id).append(option); });
          }
          $('#profile').value = 'supportive'; updateProfile();
          $('#privacyNotice').textContent = config.privacyNotice;
        }
        // A página continua apresentando os campos mesmo se a chamada ao host atrasar.
        hydrate(fallback);
        if (hasHost()) {
          $('#privacyNotice').textContent = 'Conectando ao serviço de tutoria...';
          $('#loginButton').disabled = true;
          try {
            hydrate(await host('getBootstrap'));
          } catch {
            $('#userChip').textContent = 'Serviço indisponível';
            $('#privacyNotice').textContent = 'Não foi possível carregar a configuração. Recarregue a página ou confira a versão implantada no Apps Script.';
            return;
          } finally { $('#loginButton').disabled = false; }
        }
        $('#userChip').textContent = hasHost() ? 'Entre para estudar' : 'Prévia local';
      }
      boot().catch(() => {
        $('#userChip').textContent = 'Falha ao iniciar';
        $('#privacyNotice').textContent = 'Não foi possível iniciar a interface. Confira se o arquivo Index.html foi copiado integralmente e publique uma nova versão.';
      });
      })();
    </script>
